Configurar Conexión Supabase

Ingresa las credenciales de tu proyecto Supabase para sincronizar clientes y pedidos directamente con tu base de datos en la nube.

Script de Tablas para Supabase SQL Editor

Copia y ejecuta este script en el SQL Editor de tu proyecto Supabase para crear automáticamente las tablas clientes y pedidos. 

-- 1. Tabla de Clientes
CREATE TABLE IF NOT EXISTS clientes (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    nombres TEXT NOT NULL,
    apellidos TEXT NOT NULL,
    dni TEXT NOT NULL,
    telefono TEXT NOT NULL,
    direccion TEXT NOT NULL,
    referencia TEXT NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 2. Tabla de Pedidos
CREATE TABLE IF NOT EXISTS pedidos (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    order_code TEXT NOT NULL,
    cliente_dni TEXT NOT NULL,
    cliente_nombre TEXT NOT NULL,
    direccion TEXT NOT NULL,
    items JSONB NOT NULL,
    subtotal NUMERIC(10,2) NOT NULL,
    delivery_fee NUMERIC(10,2) NOT NULL,
    monto_total NUMERIC(10,2) NOT NULL,
    metodo_pago TEXT NOT NULL,
    estado TEXT DEFAULT 'Recibido',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Deshabilitar RLS para pruebas rápidas
ALTER TABLE clientes DISABLE ROW LEVEL SECURITY;
ALTER TABLE pedidos DISABLE ROW LEVEL SECURITY;
